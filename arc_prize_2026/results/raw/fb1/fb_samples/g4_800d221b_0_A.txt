import numpy as np

def transform(grid):
    grid = [list(row) for row in grid]
    h, w = len(grid), len(grid[0])
    a = np.array(grid, dtype=int)

    # Identify the three non-background colors:
    # background = most frequent color
    # special = color that is changed in the training examples
    # object colors = the other two colors
    vals, counts = np.unique(a, return_counts=True)
    bg = int(vals[np.argmax(counts)])

    # special color is the one that is not background and has the
    # second-highest frequency among non-background colors
    non_bg = [(v, c) for v, c in zip(vals, counts) if v != bg]
    non_bg.sort(key=lambda x: -x[1])
    special = int(non_bg[0][0])
    obj1 = int(non_bg[1][0])
    obj2 = int(non_bg[2][0])

    # Build masks
    is_bg = (a == bg)
    is_special = (a == special)
    is_obj1 = (a == obj1)
    is_obj2 = (a == obj2)

    # Flood fill from each object color through special cells,
    # treating background as walls.
    def fill(start_mask):
        dist = np.full((h, w), -1, dtype=int)
        q = []
        ys, xs = np.where(start_mask)
        for y, x in zip(ys, xs):
            dist[y, x] = 0
            q.append((y, x))
        head = 0
        while head < len(q):
            y, x = q[head]
            head += 1
            d = dist[y, x]
            for dy, dx in ((1,0),(-1,0),(0,1),(0,-1)):
                ny, nx = y+dy, x+dx
                if 0 <= ny < h and 0 <= nx < w:
                    if dist[ny, nx] == -1 and not is_bg[ny, nx] and (is_special[ny, nx] or start_mask[ny, nx]):
                        dist[ny, nx] = d + 1
                        q.append((ny, nx))
        return dist

    d1 = fill(is_obj1)
    d2 = fill(is_obj2)

    out = a.copy()

    for y in range(h):
        for x in range(w):
            if is_special[y, x]:
                dd1 = d1[y, x]
                dd2 = d2[y, x]
                if dd1 == -1 and dd2 == -1:
                    # keep special if unreachable
                    out[y, x] = special
                elif dd1 == -1:
                    out[y, x] = obj2
                elif dd2 == -1:
                    out[y, x] = obj1
                elif dd1 < dd2:
                    out[y, x] = obj1
                elif dd2 < dd1:
                    out[y, x] = obj2
                else:
                    # tie: keep special
                    out[y, x] = special

    return out.tolist()
